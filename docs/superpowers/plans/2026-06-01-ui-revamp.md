# UI Revamp (Phase 6 `ui-revamp`) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Restructure KnightCode's core TUI surfaces (input box, streaming message view, tool-call/diff rendering, status bar/footer, dialogs) on the existing OpenTUI/React stack to mirror claude-code, plus a per-spawn Agent model picker — shipped as one `ui-revamp` branch / one squashed commit / one PR.

**Architecture:** Pull the testable presentation logic out of the bloated rendering components into small pure `lib/ui/` helpers (TDD'd with `bun:test`), then rebuild the visual components on top of those helpers. Tool calls render collapsed-with-status by default and expand on demand; diffs gain line-number gutters and themed add/remove colors; the input box and status bar/footer surface mode + model + reasoning + cost/context as themed chips; dialogs adopt a shared visual language. The Agent tool gains a confirmation gate whose UI lets the user override the spawn's model for that one run (raw OpenRouter id threaded through `executeLocalTool`). No new product features beyond §6.9; `useChat`, routing, providers, keyboard-layer, and theme infrastructure are preserved.

**Tech Stack:** Bun 1.3.3 (Windows/PowerShell), `@opentui/react` + `@opentui/core` (React reconciler renderer), `bun:test`, `@knightcode/shared` (models/aliases), `ai` / `@ai-sdk/react` `useChat`.

**Spec:** `docs/superpowers/specs/2026-05-30-byok-openrouter-cutover-design.md` §6.9 (UI revamp) and §6.3 (subagent model override). Fidelity reference: `claude-code/src` for layout/behavior, `opencode/` secondary.

- **OpenTUI gotchas (learned in prior phases):**
  - `backgroundColor` is a `BoxProps` prop, **not** `TextProps`. To highlight a row, wrap the `<text>` in a `<box backgroundColor=…>`.
  - Inputs: `<input ref={…} />` exposes `.value`; `<textarea ref={…} />` exposes `.plainText` / `.cursorOffset` / `.setText()` / `.replaceText()`.
  - Keyboard handlers gate on `isTopLayer(id)`; overlays `push`/`pop` their layer id in a `useEffect`.
  - Theme colors come from `useTheme().colors` (`ThemeColors`); **reuse existing keys** (`success`, `error`, `autoMode`, `info`, `thinkingBorder`, `dimSeparator`, `selection`, `surface`, `dialogSurface`) rather than adding new ones — adding a key forces editing all ~33 theme objects in `providers/theme/theme.ts`.

```ts
// packages/cli/src/lib/ui/tool-presentation.ts

export type ToolStatus = "running" | "success" | "error";

/** Glyph shown next to a tool row for each status. */
export const STATUS_GLYPH: Record<ToolStatus, string> = {
  running: "…",
  success: "✓",
  error: "✗",
};

const SUMMARY_MAX = 120;

/** "TodoWrite" → "Todo Write"; leaves single words untouched. */
export function formatToolName(name: string): string {
  return name
    .replace(/([a-z0-9])([A-Z])/g, "$1 $2")
    .replace(/^./, (c) => c.toUpperCase());
}

function truncate(text: string): string {
  const oneLine = text.replace(/\s+/g, " ").trim();
  if (oneLine.length <= SUMMARY_MAX) return oneLine;
  return oneLine.slice(0, SUMMARY_MAX - 1) + "…";
}

/** A short, single-line description of what a tool call is doing. */
export function summarizeToolInput(toolName: string, input: unknown): string {
  if (input == null || typeof input !== "object") {
    return input == null ? "" : truncate(String(input));
  }
  const obj = input as Record<string, unknown>;

  switch (toolName) {
    case "Read":
    case "Edit":
    case "MultiEdit":
    case "NotebookEdit":
      return truncate(String(obj.file_path ?? "file"));
    case "Write": {
      const len = String(obj.content ?? "").length;
      return truncate(`${String(obj.file_path ?? "file")} (${len} chars)`);
    }
    case "Bash":
      return truncate(String(obj.command ?? ""));
    case "Grep":
    case "Glob":
      return truncate(String(obj.pattern ?? ""));
    case "WebFetch":
      return truncate(String(obj.url ?? ""));
    case "WebSearch":
      return truncate(String(obj.query ?? ""));
    case "Skill":
      return truncate(String(obj.command ?? obj.name ?? ""));
    case "Config":
      return truncate(String(obj.key ?? ""));
    case "Agent": {
      const sub = obj.subagent_type ? `${String(obj.subagent_type)} — ` : "";
      return truncate(`${sub}${String(obj.description ?? "")}`);
    }
    case "TodoWrite": {
      const todos = Array.isArray(obj.todos) ? obj.todos : [];
      const completed = todos.filter(
        (t: any) => t?.status === "completed",
      ).length;
      return `checklist (${completed}/${todos.length} completed)`;
    }
    default:
      return truncate(
        Object.values(obj)
          .map((v) =>
            v == null
              ? ""
              : typeof v === "object"
                ? JSON.stringify(v)
                : String(v),
          )
          .filter(Boolean)
          .join(" "),
      );
  }
}

/** Derive a coarse status from an AI SDK tool part's `state`. */
export function toolStatus(part: { state?: string }): ToolStatus {
  if (part.state === "output-error") return "error";
  if (part.state === "output-available") return "success";
  return "running";
}
```

```ts
// packages/cli/src/lib/ui/diff-rows.ts
import { computeLineDiff } from "../git/diff";

export type DiffRowKind = "added" | "removed" | "context";

export type DiffRow = {
  kind: DiffRowKind;
  text: string;
  oldNo?: number;
  newNo?: number;
};

export type BuildDiffRowsOptions = {
  maxChars?: number;
  maxLines?: number;
};

const DEFAULT_MAX_CHARS = 10_000;
const DEFAULT_MAX_LINES = 500;

/**
 * Line-numbered diff rows from old/new strings, with a single-row truncation
 * placeholder when the change is too large to render usefully.
 */
export function buildDiffRows(
  oldString: string,
  newString: string,
  options: BuildDiffRowsOptions = {},
): { rows: DiffRow[]; truncated: boolean } {
  const maxChars = options.maxChars ?? DEFAULT_MAX_CHARS;
  const maxLines = options.maxLines ?? DEFAULT_MAX_LINES;

  const combinedChars = oldString.length + newString.length;
  const combinedLines =
    oldString.split("\n").length + newString.split("\n").length;

  if (combinedChars > maxChars || combinedLines > maxLines) {
    return {
      truncated: true,
      rows: [
        {
          kind: "context",
          text: `[Diff too large to display (${combinedChars} characters, ${combinedLines} lines)]`,
        },
      ],
    };
  }

  let oldNo = 0;
  let newNo = 0;
  const rows: DiffRow[] = computeLineDiff(oldString, newString).map((line) => {
    if (line.type === "added") {
      newNo += 1;
      return { kind: "added", text: line.content, newNo };
    }
    if (line.type === "deleted") {
      oldNo += 1;
      return { kind: "removed", text: line.content, oldNo };
    }
    oldNo += 1;
    newNo += 1;
    return { kind: "context", text: line.content, oldNo, newNo };
  });

  return { rows, truncated: false };
}
```

```tsx
// packages/cli/src/components/messages/tool-call-view.tsx
import { TextAttributes } from "@opentui/core";
import { useState } from "react";
import { useTheme } from "../../providers/theme";
import { EmptyBorder } from "../utils/border";
import {
  STATUS_GLYPH,
  formatToolName,
  summarizeToolInput,
  toolStatus,
} from "../../lib/ui/tool-presentation";

type Props = {
  toolName: string;
  input: unknown;
  state?: string;
  output?: unknown;
  errorText?: string;
  /** Start expanded (e.g. while pending confirmation). */
  defaultExpanded?: boolean;
};

function statusColor(
  status: ReturnType<typeof toolStatus>,
  colors: ReturnType<typeof useTheme>["colors"],
): string {
  if (status === "error") return colors.error;
  if (status === "success") return colors.success;
  return colors.dimSeparator;
}

function stringifyOutput(output: unknown): string {
  if (output == null) return "";
  if (typeof output === "string") return output;
  try {
    return JSON.stringify(output, null, 2);
  } catch {
    return String(output);
  }
}

export function ToolCallView({
  toolName,
  input,
  state,
  output,
  errorText,
  defaultExpanded = false,
}: Props) {
  const { colors } = useTheme();
  const [expanded, setExpanded] = useState(defaultExpanded);
  const status = toolStatus({ state });
  const summary = summarizeToolInput(toolName, input);
  const detail = errorText ?? stringifyOutput(output);
  const detailLines = detail ? detail.split("\n").slice(0, 40) : [];

  return (
    <box
      border={["left"]}
      borderColor={colors.thinkingBorder}
      customBorderChars={{ ...EmptyBorder, vertical: "│" }}
      width="100%"
      paddingX={2}
      flexDirection="column"
      onMouseDown={() => detailLines.length > 0 && setExpanded((e) => !e)}
    >
      <box flexDirection="row" gap={1} width="100%">
        <text fg={statusColor(status, colors)}>{STATUS_GLYPH[status]}</text>
        <text attributes={TextAttributes.DIM}>
          <em fg={colors.info}>{formatToolName(toolName)}</em>
        </text>
        <box flexGrow={1} flexShrink={1} overflow="hidden">
          <text attributes={TextAttributes.DIM}>{summary}</text>
        </box>
        {detailLines.length > 0 ? (
          <text attributes={TextAttributes.DIM} fg={colors.dimSeparator}>
            {expanded ? "▾" : "▸"}
          </text>
        ) : null}
      </box>

      {expanded && detailLines.length > 0 ? (
        <box flexDirection="column" paddingTop={0} width="100%">
          {detailLines.map((line, i) => (
            <text
              key={i}
              fg={status === "error" ? colors.error : colors.dimSeparator}
            >
              {line || " "}
            </text>
          ))}
        </box>
      ) : null}
    </box>
  );
}
```

```tsx
// packages/cli/src/components/messages/diff-view.tsx
import { TextAttributes } from "@opentui/core";
import { useTheme } from "../../providers/theme";
import { buildDiffRows } from "../../lib/ui/diff-rows";

type Props = {
  filePath: string;
  oldString: string;
  newString: string;
  pending?: boolean;
  errorText?: string;
};

function gutter(n?: number): string {
  return (n == null ? "" : String(n)).padStart(4, " ");
}

export function DiffView({
  filePath,
  oldString,
  newString,
  pending = false,
  errorText,
}: Props) {
  const { colors } = useTheme();
  const { rows } = buildDiffRows(oldString, newString);

  return (
    <box
      border={["top", "bottom", "left", "right"]}
      borderColor={pending ? colors.autoMode : colors.thinkingBorder}
      padding={1}
      flexDirection="column"
      width="100%"
      marginY={1}
    >
      <box flexDirection="row" justifyContent="space-between" width="100%">
        <text>{filePath || "file"}</text>
        {errorText ? <text fg={colors.error}>Failed: {errorText}</text> : null}
      </box>

      <box flexDirection="column" marginTop={1} width="100%">
        {rows.map((row, i) => {
          const fg =
            row.kind === "added"
              ? colors.success
              : row.kind === "removed"
                ? colors.error
                : colors.dimSeparator;
          const sign =
            row.kind === "added" ? "+" : row.kind === "removed" ? "-" : " ";
          return (
            <text key={i} fg={fg}>
              {gutter(row.oldNo)} {gutter(row.newNo)} {sign} {row.text}
            </text>
          );
        })}
      </box>

      {pending ? (
        <box marginTop={1}>
          <text fg={colors.autoMode} attributes={TextAttributes.BOLD}>
            Accept changes? [y] Yes [n] No [a] Always
          </text>
        </box>
      ) : null}
    </box>
  );
}
```

```ts
// packages/cli/src/lib/ui/status-format.ts

export type TokenStats = {
  inputTokens: number;
  outputTokens: number;
  totalCost: number;
  lastInputTokens?: number;
};

export type ContextSeverity = "ok" | "warn" | "crit";

/** "$0.0123 (1.5k tkn)" / "Free (2.0k tkn)" / null when there's nothing yet. */
export function formatTokenCost(
  stats: Pick<TokenStats, "inputTokens" | "outputTokens" | "totalCost">,
): string | null {
  if (stats.inputTokens <= 0 && stats.outputTokens <= 0) return null;
  const totalK = ((stats.inputTokens + stats.outputTokens) / 1000).toFixed(1);
  const money = stats.totalCost > 0 ? `$${stats.totalCost.toFixed(4)}` : "Free";
  return `${money} (${totalK}k tkn)`;
}

export function formatContextWindow(
  lastInputTokens: number | undefined,
  contextLimit: number,
): {
  percentLeft: number;
  remainingK: string;
  limitK: string;
  severity: ContextSeverity;
} | null {
  if (lastInputTokens === undefined) return null;
  const remaining = Math.max(0, contextLimit - lastInputTokens);
  const percentLeft = Math.round((remaining / contextLimit) * 100);
  const severity: ContextSeverity =
    percentLeft <= 30 ? "crit" : percentLeft <= 50 ? "warn" : "ok";
  return {
    percentLeft,
    remainingK: (remaining / 1000).toFixed(0),
    limitK: (contextLimit / 1000).toFixed(0),
    severity,
  };
}
```

```ts
// packages/cli/src/lib/inference/resolve-subagent-model.ts
import {
  DEFAULT_CHAT_MODEL_ID,
  findSupportedChatModel,
  resolveModelAlias,
  type SupportedChatModelId,
} from "@knightcode/shared";

export type ResolveSubagentModelArgs = {
  /** Raw OpenRouter id chosen by the user at spawn-confirm time. */
  override?: string;
  /** The Agent tool's `model` alias arg (model-chosen). */
  aliasArg?: string;
  /** The agent definition's `model` ("inherit" | alias | undefined). */
  agentModel?: string;
};

/**
 * Precedence: user per-spawn override (raw id, must be a known model) >
 * tool alias arg > agent default alias (unless "inherit") > global default.
 */
export function resolveSubagentModel(
  args: ResolveSubagentModelArgs,
): SupportedChatModelId {
  if (args.override && findSupportedChatModel(args.override)) {
    return args.override as SupportedChatModelId;
  }
  const fromAlias = args.aliasArg && resolveModelAlias(args.aliasArg);
  if (fromAlias) return fromAlias;
  const fromAgent =
    args.agentModel &&
    args.agentModel !== "inherit" &&
    resolveModelAlias(args.agentModel);
  if (fromAgent) return fromAgent;
  return DEFAULT_CHAT_MODEL_ID;
}
```
